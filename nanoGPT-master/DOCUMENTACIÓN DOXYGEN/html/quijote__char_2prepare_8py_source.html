1"""
2Prepare the Shakespeare dataset for character-level language modeling.
3So instead of encoding with GPT-2 BPE tokens, we just map characters to ints.
4Will save train.bin, val.bin containing the ids, and meta.pkl containing the
5encoder and decoder and some other related info.
6"""
7import os
8import pickle
9import numpy as np

11# This dataset is expected to be provided locally as input.txt.

13if not os.path.exists(input_file_path):
14 raise FileNotFoundError(
15 f"Missing {input_file_path}. Put the Quijote text in this file before running prepare.py."
16 )
17
18with open(input_file_path, 'r', encoding='utf-8') as f:
19 data = f.read()
20print(f"length of dataset in characters: {len(data):,}")

29"""
30# Conversión del dataset Shakespeare de letras a números.
31
32**Autor:** Andrej Karpathy
33**Documentación en español:** *Josué Aybar Paulino*
34
35## Descripción
36
37`Prepare.py` cuenta con su propio tokenizador que le permite convertir de letras a números. Se auxilia de la librería
38de Python `stoi` para esta conversión. Luego, los datos se almacenan en `train.bil` y `val.bin`. Es en train.bil donde
39los datos son leídos y se crean las dos secuencias de números X y Y, donde Y está desplazada a una posición de X.
40Finalmente, la librería itos convierte esos datos numéricos a letras.
41"""
42# create a mapping from characters to integers
43stoi = { ch:i for i,ch in enumerate(chars) }
44itos = { i:ch for i,ch in enumerate(chars) }
45def encode(s):
46 return [stoi[c] for c in s] # encoder: take a string, output a list of integers
47def decode(l):
48 return ''.join([itos[i] for i in l]) # decoder: take a list of integers, output a string
49
50# create the train and test splits
51n = len(data)
52train_data = data[:int(n*0.9)]
53val_data = data[int(n*0.9):]
54
55# encode both to integers
56train_ids = encode(train_data)
57val_ids = encode(val_data)
58print(f"train has {len(train_ids):,} tokens")
59print(f"val has {len(val_ids):,} tokens")
60
61# export to bin files
62train_ids = np.array(train_ids, dtype=np.uint16)
63val_ids = np.array(val_ids, dtype=np.uint16)
64train_ids.tofile(os.path.join(os.path.dirname(__file__), 'train.bin'))
65val_ids.tofile(os.path.join(os.path.dirname(__file__), 'val.bin'))
66
67# save the meta information as well, to help us encode/decode later
68meta = {
69 'vocab_size': vocab_size,
70 'itos': itos,
71 'stoi': stoi,
72}
73with open(os.path.join(os.path.dirname(__file__), 'meta.pkl'), 'wb') as f:
74 pickle.dump(meta, f)
