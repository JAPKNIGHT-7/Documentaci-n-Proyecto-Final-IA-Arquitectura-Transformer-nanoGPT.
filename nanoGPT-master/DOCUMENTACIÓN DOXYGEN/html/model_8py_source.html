17class LayerNorm(nn.Module):
18 """
19 @brief Capa encargada de normalizar las representaciones internas del modelo.
20
21 @author Andrej Karpathy
22 **Documentación en español:** *Josué Aybar Paulino*
23
24 @details
25
26 La clase **LayerNorm** es una capa que normaliza los valores internos del modelo para mantenerlos en una escala estable,
27 facilitando el entrenamiento de los bloques Transformer. La función `__init__` crea los parámetros entrenables `weight`
28 y `bias` según la dimensión del embedding (ndim), que en este modelo corresponde a 768 dimensiones. También permite
29 definir si se utilizará el término de sesgo. La función `super()` inicializa la clase base de PyTorch para que esta
30 capa pueda ser gestionada como un componente entrenable. Finalmente, `forward()` recibe los datos numéricos de entrada
31 y aplica la normalización mediante layer_norm antes de continuar el procesamiento dentro del Transformer.
32 """
33
34 def __init__(self, ndim, bias):
35 super().__init__()
36 self.weight = nn.Parameter(torch.ones(ndim))
37 self.bias = nn.Parameter(torch.zeros(ndim)) if bias else None
38
39 def forward(self, input):
40 return F.layer_norm(input, self.weight.shape, self.weight, self.bias, 1e-5)

43class CausalSelfAttention(nn.Module):
44 """
45 @brief Implementación del mecanismo de atención causal utilizado por los bloques Transformer de GPT.
46
47 @author Andrej Karpathy
48 **Documentación en español:** *Josué Aybar Paulino*
49
50 @details
51
52 La clase **CausalSelfAttention** implementa el mecanismo de atención causal de los bloques Transformer del modelo GPT.
53 Su función es analizar las relaciones entre los tokens utilizando únicamente la información de los tokens anteriores
54 para comprender el contexto de la secuencia. La función `__init__()` inicializa las capas y parámetros necesarios para
55 el mecanismo de atención, mientras que `forward()` calcula la atención y genera una nueva representación contextualizada
56 de los tokens.
57 """
58
59 def __init__(self, config):
60 super().__init__()
61 assert config.n_embd % config.n_head == 0
62
63 self.c_attn = nn.Linear(config.n_embd, 3 * config.n_embd, bias=config.bias)
64
65 self.c_proj = nn.Linear(config.n_embd, config.n_embd, bias=config.bias)
66
67 self.attn_dropout = nn.Dropout(config.dropout)
68 self.resid_dropout = nn.Dropout(config.dropout)
69 self.n_head = config.n_head
70 self.n_embd = config.n_embd
71 self.dropout = config.dropout
72
73 self.flash = hasattr(torch.nn.functional, 'scaled_dot_product_attention')
74 if not self.flash:
75 print("WARNING: using slow attention. Flash Attention requires PyTorch >= 2.0")
76
77 self.register_buffer("bias", torch.tril(torch.ones(config.block_size, config.block_size))
78 .view(1, 1, config.block_size, config.block_size))
79
80 def forward(self, x):
81 B, T, C = x.size()
82
83
84 q, k, v = self.c_attn(x).split(self.n_embd, dim=2)
85 k = k.view(B, T, self.n_head, C // self.n_head).transpose(1, 2) # (B, nh, T, hs)
86 q = q.view(B, T, self.n_head, C // self.n_head).transpose(1, 2) # (B, nh, T, hs)
87 v = v.view(B, T, self.n_head, C // self.n_head).transpose(1, 2) # (B, nh, T, hs)
88
89
90 if self.flash:
91
92 y = torch.nn.functional.scaled_dot_product_attention(q, k, v, attn_mask=None, dropout_p=self.dropout if self.training else 0, is_causal=True)
93 else:
94
95 att = (q @ k.transpose(-2, -1)) * (1.0 / math.sqrt(k.size(-1)))
96 att = att.masked_fill(self.bias[:,:,:T,:T] == 0, float('-inf'))
97 att = F.softmax(att, dim=-1)
98 att = self.attn_dropout(att)
99 y = att @ v # (B, nh, T, T) x (B, nh, T, hs) -> (B, nh, T, hs)
100 y = y.transpose(1, 2).contiguous().view(B, T, C) # re-assemble all head outputs side by side
101
102
103 y = self.resid_dropout(self.c_proj(y))
104 return y

107class MLP(nn.Module):
108 """
109 @brief Implementación del Perceptrón Multicapa (MLP) de un bloque Transformer.
110
111 @author Andrej Karpathy
112 **Documentación en español:** *Josué Aybar Paulino*
113
114 @details
115
116 Procesa la representación generada por `CausalSelfAttention` mediante dos capas lineales, la función
117 de activación `GELU` y `Dropout`, generando una nueva representación del token que será enviada al
118 siguiente bloque Transformer.
119
120 @param config Configuración de la arquitectura GPT.
121 """
122 def __init__(self, config):
123 super().__init__()
124 self.c_fc = nn.Linear(config.n_embd, 4 * config.n_embd, bias=config.bias)
125 self.gelu = nn.GELU()
126 self.c_proj = nn.Linear(4 * config.n_embd, config.n_embd, bias=config.bias)
127 self.dropout = nn.Dropout(config.dropout)
128
129 def forward(self, x):
130 x = self.c_fc(x)
131 x = self.gelu(x)
132 x = self.c_proj(x)
133 x = self.dropout(x)
134 return x

137class Block(nn.Module):
138 """
139 @brief Bloque Transformer de GPT.
140
141 @author Andrej Karpathy
142 **Documentación en español:** *Josué Aybar Paulino*
143
144 @details
145
146 Integra los componentes principales de un bloque Transformer:
147 - `LayerNorm`: normalización para estabilizar el entrenamiento.
148 - `CausalSelfAttention`: atención causal sobre el contexto previo.
149 - `MLP`: transformación no lineal de las representaciones.
150
151 ## Flujo
152 `__init__()` inicializa los componentes.
153 `forward()` aplica normalización, atención causal, MLP y conexiones residuales.
154 """
155 def __init__(self, config):
156 super().__init__()
157 self.ln_1 = LayerNorm(config.n_embd, bias=config.bias)
158 self.attn = CausalSelfAttention(config)
159 self.ln_2 = LayerNorm(config.n_embd, bias=config.bias)
160 self.mlp = MLP(config)
161
162 def forward(self, x):
163 x = x + self.attn(self.ln_1(x))
164 x = x + self.mlp(self.ln_2(x))
165 return x

197class GPT(nn.Module):
198 """
199 @brief Modelo `GPT` completo basado en arquitectura Transformer.
200
201 @author Andrej Karpathy
202 **Documentación en español:** *Josué Aybar Paulino*
203
204 @details
205
206 Integra los componentes principales del modelo Transformer:
207
208 - **Embeddings de tokens y posiciones**.
209 - **Dropout** para regularización.
210 - **Bloques Transformer**, compuestos por autoatención y MLP.
211 - **LayerNorm final** para normalización.
212 - **`lm_head`**: convierte la representación final en *logits* para predecir el siguiente token.
213
214 @warning Esta clase integra todos los componentes del modelo GPT y coordina su ejecución de forma unificada.
215 """
216 ## Inicialización de la arquitectura del modelo GPT.
217 def __init__(self, config):
218 super().__init__()
219 assert config.vocab_size is not None
220 assert config.block_size is not None
221 self.config = config
222
223 self.transformer = nn.ModuleDict(dict(
224 wte = nn.Embedding(config.vocab_size, config.n_embd),
225 wpe = nn.Embedding(config.block_size, config.n_embd),
226 drop = nn.Dropout(config.dropout),
227 h = nn.ModuleList([Block(config) for _ in range(config.n_layer)]),
228 ln_f = LayerNorm(config.n_embd, bias=config.bias),
229 ))
230 self.lm_head = nn.Linear(config.n_embd, config.vocab_size, bias=False)
231 ## with weight tying when using torch.compile() some warnings get generated:
232 ## "UserWarning: functional_call was passed multiple values for tied weights.
233 ## This behavior is deprecated and will be an error in future versions"
234 ## not 100% sure what this is, so far seems to be harmless. @TODO investigate
235 self.transformer.wte.weight = self.lm_head.weight ## https://paperswithcode.com/method/weight-tying
236
237 ## init all weights
238 self.apply(self._init_weights)
239 ## apply special scaled init to the residual projections, per GPT-2 paper
240 for pn, p in self.named_parameters():
241 if pn.endswith('c_proj.weight'):
242 torch.nn.init.normal_(p, mean=0.0, std=0.02/math.sqrt(2 * config.n_layer))
243
244 ## report number of parameters
245 print("number of parameters: %.2fM" % (self.get_num_params()/1e6,))

285 def forward(self, idx, targets=None):
286 """
287 @brief Procesamiento de la secuencia de entrada.
288
289 @author Andrej Karpathy
290 **Documentación en español:** *Josué Aybar Paulino*
291
292 @details
293
294 Ejecuta la propagación hacia adelante (*forward pass*) del modelo GPT.
295 Convierte los tokens en embeddings, procesa la secuencia mediante los
296 bloques Transformer y genera los logits utilizados para predecir el
297 siguiente token.
298
299 @param idx Tensor que contiene los índices de los tokens de entrada.
300 @param targets Tensor con los tokens objetivo utilizados para calcular
301 la pérdida durante el entrenamiento. Puede ser None durante la inferencia.
302
303 @return Una tupla `(logits, loss)` con los logits de salida y la pérdida.
304 """
305 device = idx.device
306 b, t = idx.size()
307 assert t <= self.config.block_size, f"Cannot forward sequence of length {t}, block size is only {self.config.block_size}"
308 pos = torch.arange(0, t, dtype=torch.long, device=device) # shape (t)
309
310 ## forward the GPT model itself
311 tok_emb = self.transformer.wte(idx) ## token embeddings of shape (b, t, n_embd)
312 pos_emb = self.transformer.wpe(pos) ## position embeddings of shape (t, n_embd)
313 x = self.transformer.drop(tok_emb + pos_emb)
314 for block in self.transformer.h:
315 x = block(x)
316 x = self.transformer.ln_f(x)
317
318 if targets is not None:
319 ## if we are given some desired targets also calculate the loss
320 logits = self.lm_head(x)
321 loss = F.cross_entropy(logits.view(-1, logits.size(-1)), targets.view(-1), ignore_index=-1)
322 else:
323 ## inference-time mini-optimization: only forward the lm_head on the very last position
324 logits = self.lm_head(x[:, [-1], :]) ## note: using list [-1] to preserve the time dim
325 loss = None
326
327 return logits, loss

343 @classmethod
344 def from_pretrained(cls, model_type, override_args=None):
345 """
346 @brief Carga de un modelo GPT preentrenado.
347
348 @author Andrej Karpathy
349 **Documentación en español:** *Josué Aybar Paulino*
350
351 @details
352
353 Crea una instancia del modelo e importa los pesos preentrenados de GPT-2 desde Hugging Face.
354 """
355 assert model_type in {'gpt2', 'gpt2-medium', 'gpt2-large', 'gpt2-xl'}
356 override_args = override_args or {} # default to empty dict
357 ## only dropout can be overridden see more notes below
358 assert all(k == 'dropout' for k in override_args)
359 from transformers import GPT2LMHeadModel
360 print("loading weights from pretrained gpt: %s" % model_type)
361
362 ## n_layer, n_head and n_embd are determined from model_type
363 config_args = {
364 'gpt2': dict(n_layer=12, n_head=12, n_embd=768), # 124M params
365 'gpt2-medium': dict(n_layer=24, n_head=16, n_embd=1024), # 350M params
366 'gpt2-large': dict(n_layer=36, n_head=20, n_embd=1280), # 774M params
367 'gpt2-xl': dict(n_layer=48, n_head=25, n_embd=1600), # 1558M params
368 }[model_type]
369 print("forcing vocab_size=50257, block_size=1024, bias=True")
370 config_args['vocab_size'] = 50257 ## always 50257 for GPT model checkpoints
371 config_args['block_size'] = 1024 ## always 1024 for GPT model checkpoints
372 config_args['bias'] = True ## always True for GPT model checkpoints
373 ## we can override the dropout rate, if desired
374 if 'dropout' in override_args:
375 print(f"overriding dropout rate to {override_args['dropout']}")
376 config_args['dropout'] = override_args['dropout']
377 ## create a from-scratch initialized minGPT model
378 config = GPTConfig(**config_args)
379 model = GPT(config)
380 sd = model.state_dict()
381 sd_keys = sd.keys()
382 sd_keys = [k for k in sd_keys if not k.endswith('.attn.bias')] # discard this mask / buffer, not a param
383
384 ## init a huggingface/transformers model
385 model_hf = GPT2LMHeadModel.from_pretrained(model_type)
386 sd_hf = model_hf.state_dict()
387
388 ## copy while ensuring all of the parameters are aligned and match in names and shapes
389 sd_keys_hf = sd_hf.keys()
390 sd_keys_hf = [k for k in sd_keys_hf if not k.endswith('.attn.masked_bias')] # ignore these, just a buffer
391 sd_keys_hf = [k for k in sd_keys_hf if not k.endswith('.attn.bias')] # same, just the mask (buffer)
392 transposed = ['attn.c_attn.weight', 'attn.c_proj.weight', 'mlp.c_fc.weight', 'mlp.c_proj.weight']
393 ## basically the openai checkpoints use a "Conv1D" module, but we only want to use a vanilla Linear
394 ## this means that we have to transpose these weights when we import them
395 assert len(sd_keys_hf) == len(sd_keys), f"mismatched keys: {len(sd_keys_hf)} != {len(sd_keys)}"
396 for k in sd_keys_hf:
397 if any(k.endswith(w) for w in transposed):
398 ## special treatment for the Conv1D weights we need to transpose
399 assert sd_hf[k].shape[::-1] == sd[k].shape
400 with torch.no_grad():
401 sd[k].copy_(sd_hf[k].t())
402 else:
403 ## vanilla copy over the other parameters
404 assert sd_hf[k].shape == sd[k].shape
405 with torch.no_grad():
406 sd[k].copy_(sd_hf[k])
407
408 return model

411 def configure_optimizers(self, weight_decay, learning_rate, betas, device_type):
412 """
413 @brief Configura el optimizador AdamW para el entrenamiento.
414
415 @author Andrej Karpathy
416 **Documentación en español:** *Josué Aybar Paulino*
417
418 @details
419
420 Organiza los parámetros del modelo en grupos con y sin weight decay
421 y configura AdamW con la tasa de aprendizaje y los coeficientes beta.
422
423 @param weight_decay Factor de regularización.
424 @param learning_rate Tasa de aprendizaje.
425 @param betas Coeficientes beta de AdamW.
426 @param device_type Tipo de dispositivo utilizado.
427
428 @return Optimizador AdamW configurado.
429 """
430 ## start with all of the candidate parameters
431 param_dict = {pn: p for pn, p in self.named_parameters()}
432 ## filter out those that do not require grad
433 param_dict = {pn: p for pn, p in param_dict.items() if p.requires_grad}
434 ## create optim groups. Any parameters that is 2D will be weight decayed, otherwise no.
435 ## i.e. all weight tensors in matmuls + embeddings decay, all biases and layernorms don't.
436 decay_params = [p for n, p in param_dict.items() if p.dim() >= 2]
437 nodecay_params = [p for n, p in param_dict.items() if p.dim() < 2]
438 optim_groups = [
439 {'params': decay_params, 'weight_decay': weight_decay},
440 {'params': nodecay_params, 'weight_decay': 0.0}
441 ]
442 num_decay_params = sum(p.numel() for p in decay_params)
443 num_nodecay_params = sum(p.numel() for p in nodecay_params)
444 print(f"num decayed parameter tensors: {len(decay_params)}, with {num_decay_params:,} parameters")
445 print(f"num non-decayed parameter tensors: {len(nodecay_params)}, with {num_nodecay_params:,} parameters")
446 ## Create AdamW optimizer and use the fused version if it is available
447 fused_available = 'fused' in inspect.signature(torch.optim.AdamW).parameters
448 use_fused = fused_available and device_type == 'cuda'
449 extra_args = dict(fused=True) if use_fused else dict()
450 optimizer = torch.optim.AdamW(optim_groups, lr=learning_rate, betas=betas, **extra_args)
451 print(f"using fused AdamW: {use_fused}")
452
453 return optimizer

457 def estimate_mfu(self, fwdbwd_per_iter, dt):
458 """ estimate model flops utilization (MFU) in units of A100 bfloat16 peak FLOPS """
459 ## first estimate the number of flops we do per iteration.
460 ## see PaLM paper Appendix B as ref: https://arxiv.org/abs/2204.02311
461 N = self.get_num_params()
462 cfg = self.config
463 L, H, Q, T = cfg.n_layer, cfg.n_head, cfg.n_embd//cfg.n_head, cfg.block_size
464 flops_per_token = 6*N + 12*L*H*Q*T
465 flops_per_fwdbwd = flops_per_token * T
466 flops_per_iter = flops_per_fwdbwd * fwdbwd_per_iter
467 ## express our flops throughput as ratio of A100 bfloat16 peak flops
468 flops_achieved = flops_per_iter * (1.0/dt) ## per second
469 flops_promised = 312e12 ## A100 GPU bfloat16 peak flops is 312 TFLOPS
470 mfu = flops_achieved / flops_promised
471 return mfu

474 @torch.no_grad()
475 def generate(self, idx, max_new_tokens, temperature=1.0, top_k=None):
476 """
477 @brief Generación autoregresiva de texto.
478
479 @author Andrej Karpathy
480 **Documentación en español:** *Josué Aybar Paulino*
481
482 @details
483
484 Genera texto prediciendo un token a partir del contexto disponible.
485 En cada iteración, los logits se convierten en probabilidades mediante
486 softmax y se selecciona el siguiente token mediante muestreo.
487
488 @param idx Secuencia inicial de tokens.
489 @param max_new_tokens Número máximo de tokens nuevos que se generarán.
490 @param temperature Controla la distribución de probabilidad durante el muestreo.
491 @param top_k Limita el muestreo a los `k` tokens con mayor probabilidad.
492
493 @return Secuencia original junto con los tokens generados.
494 """
495
496 for _ in range(max_new_tokens):
497 ## if the sequence context is growing too long we must crop it at block_size
498 idx_cond = idx if idx.size(1) <= self.config.block_size else idx[:, -self.config.block_size:]
499 ## forward the model to get the logits for the index in the sequence
500 logits, _ = self(idx_cond)
501 ## pluck the logits at the final step and scale by desired temperature
502 logits = logits[:, -1, :] / temperature
503 ## optionally crop the logits to only the top k options
504 if top_k is not None:
505 v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
506 logits[logits < v[:, [-1]]] = -float('Inf')
507 ## apply softmax to convert logits to (normalized) probabilities
508 probs = F.softmax(logits, dim=-1)
509 ## sample from the distribution
510 idx_next = torch.multinomial(probs, num_samples=1)
511 ## append sampled index to the running sequence and continue
512 idx = torch.cat((idx, idx_next), dim=1)
513
514 return idx
