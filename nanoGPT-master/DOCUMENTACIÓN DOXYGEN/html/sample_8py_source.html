94def generate_visual(model, idx, max_new_tokens, temperature=1.0, top_k=None,
95 show_probs=True, top_k_display=4):
96 """
97 @brief Genera texto mostrando las probabilidades de los candidatos.
98
99 @details
100 Reescribe el proceso de generación de model.py para mostrar los candidatos
101 más probables y sus probabilidades antes de seleccionar el siguiente carácter.
102
103 @param model Modelo GPT entrenado.
104 @param idx Secuencia inicial de tokens.
105 @param max_new_tokens Número de tokens a generar.
106 @param temperature Temperatura de generación.
107 @param top_k Candidatos considerados durante el muestreo.
108 @param show_probs Muestra u oculta las probabilidades.
109 @param top_k_display Número de candidatos mostrados.
110
111 @return Secuencia original junto con los tokens generados.
112 """
113 print(f"\nContexto inicial: {decode(idx[0].tolist())!r}\n")
114
115 for paso in range(max_new_tokens):
116 # recortar el contexto al tamano maximo que soporta el modelo (block_size)
117 idx_cond = idx if idx.size(1) <= model.config.block_size else idx[:, -model.config.block_size:]
118
119 # --- pedirle al modelo los logits (esto es lo mismo que hace generate() internamente) ---
120 with torch.no_grad():
121 with ctx:
122 logits, _ = model(idx_cond)
123
124 logits = logits[:, -1, :] / temperature
125
126 # top_k opcional para el muestreo real (igual que en el generate() original)
127 if top_k is not None:
128 v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
129 logits[logits < v[:, [-1]]] = -float('Inf')
130
131 probs = F.softmax(logits, dim=-1)
132
133 # --- aqui esta la parte visual ---
134 if show_probs:
135 top_probs, top_ids = torch.topk(probs, k=top_k_display)
136 candidatos = []
137 for p, tid in zip(top_probs[0], top_ids[0]):
138 caracter = decode([tid.item()])
139 candidatos.append(f"{caracter!r} {p.item()*100:.1f}%")
140 print(f"Paso {paso+1:3d} | candidatos: " + " | ".join(candidatos))
141
142 # elegir el siguiente caracter (muestreo real, con las probs completas)
143 idx_next = torch.multinomial(probs, num_samples=1)
144 idx = torch.cat((idx, idx_next), dim=1)
145
146 # --- mostrar el caracter elegido, para que el texto se vea "escribirse" en vivo ---
147 if show_probs:
148 elegido = decode([idx_next.item()])
149 print(f" -> elegido: {elegido!r} (texto hasta ahora: {decode(idx[0].tolist())!r})\n")
150
151 return idx
152
