59def calcular_atencion(x, layer):
60 """
61 @brief Calcula la matriz de atención causal de una capa.
62
63 @author Andrej Karpathy
64 **Documentación en español:** *Josué Aybar Paulino*
65
66 @details
67
68 Recalcula externamente las puntuaciones de atención para todas
69 las cabezas y aplica la máscara causal y softmax.
70
71 @param x Entrada de la capa de atención.
72 @param layer Índice de la capa Transformer.
73
74 @return Matriz de atención de todas las cabezas en CPU.
75 """
76 attn = model.transformer.h[layer].attn
77 B, T, C = x.shape
78 q, k, _ = attn.c_attn(x).split(C, dim=2)
79 hs = C // attn.n_head
80 q = q.view(B, T, attn.n_head, hs).transpose(1, 2)
81 k = k.view(B, T, attn.n_head, hs).transpose(1, 2)
82 scores = (q @ k.transpose(-2, -1)) / (hs ** 0.5)
83 mascara = torch.tril(torch.ones(T, T, device=x.device))
84 scores = scores.masked_fill(mascara == 0, float("-inf"))
85 return torch.softmax(scores, dim=-1)[0].cpu()
