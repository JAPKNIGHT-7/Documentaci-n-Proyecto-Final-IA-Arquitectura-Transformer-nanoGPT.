3and also in a larger training run with distributed data parallel (ddp).
4
5To run on a single GPU, example:
6$ python train.py --batch_size=32 --compile=False

118def get_batch(split):
119 """
120 @brief Carga un lote de datos para entrenamiento o validación.
121
122 @author Andrej Karpathy
123 **Documentación en español:** *Josué Aybar Paulino*
124
125 @details
126
127 Carga datos desde `train.bin` o `val.bin` y genera pares de secuencias:
128 `x` contiene los tokens de entrada y `y` contiene los mismos tokens
129 desplazados una posición para predecir el siguiente token.
130
131 @param split Conjunto de datos: `train` o `val`.
132 @return Una tupla `(x, y)` con las entradas y objetivos.
133 """
134 # We recreate np.memmap every batch to avoid a memory leak, as per
135 # https://stackoverflow.com/questions/45132940/numpy-memmap-memory-usage-want-to-iterate-once/61472122#61472122
136 if split == 'train':
137 data = np.memmap(os.path.join(data_dir, 'train.bin'), dtype=np.uint16, mode='r')
138 else:
139 data = np.memmap(os.path.join(data_dir, 'val.bin'), dtype=np.uint16, mode='r')
140 ix = torch.randint(len(data) - block_size, (batch_size,))
141 x = torch.stack([torch.from_numpy((data[i:i+block_size]).astype(np.int64)) for i in ix])
142 y = torch.stack([torch.from_numpy((data[i+1:i+1+block_size]).astype(np.int64)) for i in ix])
143 if device_type == 'cuda':
144 # pin arrays x,y, which allows us to move them to GPU asynchronously (non_blocking=True)
145 x, y = x.pin_memory().to(device, non_blocking=True), y.pin_memory().to(device, non_blocking=True)
146 else:
147 x, y = x.to(device), y.to(device)
148 return x, y

263# learning rate decay scheduler (cosine with warmup)
264def get_lr(it):
265 # 1) linear warmup for warmup_iters steps
266 if it < warmup_iters:
267 return learning_rate * (it + 1) / (warmup_iters + 1)
268 # 2) if it > lr_decay_iters, return min learning rate
269 if it > lr_decay_iters:
270 return min_lr
271 # 3) in between, use cosine decay down to min learning rate
272 decay_ratio = (it - warmup_iters) / (lr_decay_iters - warmup_iters)
273 assert 0 <= decay_ratio <= 1
274 coeff = 0.5 * (1.0 + math.cos(math.pi * decay_ratio)) # coeff ranges 0..1
275 return min_lr + coeff * (learning_rate - min_lr)
