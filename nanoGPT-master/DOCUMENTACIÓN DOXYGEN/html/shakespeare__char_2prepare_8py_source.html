1"""
2Prepare the Shakespeare dataset for character-level language modeling.
3So instead of encoding with GPT-2 BPE tokens, we just map characters to ints.
4Will save train.bin, val.bin containing the ids, and meta.pkl containing the
5encoder and decoder and some other related info.
6"""
7import os
8import pickle
9import requests
10import numpy as np
11
12# download the tiny shakespeare dataset
13input_file_path = os.path.join(os.path.dirname(__file__), 'input.txt')
14if not os.path.exists(input_file_path):
15 data_url = 'https://raw.githubusercontent.com/karpathy/char-rnn/master/data/tinyshakespeare/input.txt'
16 with open(input_file_path, 'w') as f:
17 f.write(requests.get(data_url).text)
18
19with open(input_file_path, 'r') as f:
20 data = f.read()
21print(f"length of dataset in characters: {len(data):,}")
22
23# get all the unique characters that occur in this text
24chars = sorted(list(set(data)))
25vocab_size = len(chars)
26print("all the unique characters:", ''.join(chars))
27print(f"vocab size: {vocab_size:,}")
28
29
30"""
31# Conversión del dataset Shakespeare de letras a números.
32
33**Autor:** Andrej Karpathy
34**Documentación en español:** *Josué Aybar Paulino*
35
36## Descripción
37
38`Prepare.py` cuenta con su propio tokenizador que le permite convertir de letras a números. Se auxilia de la librería
39de Python `stoi` para esta conversión. Luego, los datos se almacenan en `train.bil` y `val.bin`. Es en train.bil donde
40los datos son leídos y se crean las dos secuencias de números X y Y, donde Y está desplazada a una posición de X.
41Finalmente, la librería itos convierte esos datos numéricos a letras.
42"""
43# create a mapping from characters to integers
44stoi = { ch:i for i,ch in enumerate(chars) }
45itos = { i:ch for i,ch in enumerate(chars) }
46def encode(s):
47 return [stoi[c] for c in s] # encoder: take a string, output a list of integers
48def decode(l):
49 return ''.join([itos[i] for i in l]) # decoder: take a list of integers, output a string
50
51# create the train and test splits
52n = len(data)
53train_data = data[:int(n*0.9)]
54val_data = data[int(n*0.9):]
55
56# encode both to integers
57train_ids = encode(train_data)
58val_ids = encode(val_data)
59print(f"train has {len(train_ids):,} tokens")
60print(f"val has {len(val_ids):,} tokens")
61
62# export to bin files
63train_ids = np.array(train_ids, dtype=np.uint16)
64val_ids = np.array(val_ids, dtype=np.uint16)
65train_ids.tofile(os.path.join(os.path.dirname(__file__), 'train.bin'))
66val_ids.tofile(os.path.join(os.path.dirname(__file__), 'val.bin'))
67
68# save the meta information as well, to help us encode/decode later
69meta = {
70 'vocab_size': vocab_size,
71 'itos': itos,
72 'stoi': stoi,
73}
74with open(os.path.join(os.path.dirname(__file__), 'meta.pkl'), 'wb') as f:
75 pickle.dump(meta, f)
