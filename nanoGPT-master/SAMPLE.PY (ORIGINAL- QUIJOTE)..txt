"""
Sample from a trained model
"""
import os
import pickle
from contextlib import nullcontext
import torch
import tiktoken
from model import GPTConfig, GPT

"""
# Configuración previa.

**Autor:** Andrej Karpathy
**Documentación en español:** *Josué Aybar Paulino*

## Descripción

Estas variables establecen los parámetros qe el modelo va a utilizar antes de comenzar a generar tokens.
"""
# -----------------------------------------------------------------------------
init_from = 'scratch' # 'scratch' (before training), 'resume' (from an out_dir), or a gpt2 variant
out_dir = 'out-quijote-char' # ignored if init_from is not 'resume'
dataset = 'quijote_char' # used when init_from='scratch'
start = "Érase una vez"
num_samples = 10 # number of samples to draw
max_new_tokens = 500 # number of tokens generated in each sample
temperature = 0.8 # 1.0 = no change, < 1.0 = less random, > 1.0 = more random, in predictions
top_k = 200 # retain only the top_k most likely tokens, clamp others to have 0 probability
seed = 1337
device = 'cuda' if torch.cuda.is_available() else 'cpu' # examples: 'cpu', 'cuda', 'cuda:0', 'cuda:1', etc.
dtype = 'bfloat16' if torch.cuda.is_available() and torch.cuda.is_bf16_supported() else 'float16' # 'float32' or 'bfloat16' or 'float16'
compile = False # use PyTorch 2.0 to compile the model to be faster
bias = False # do we use bias inside LayerNorm and Linear layers?
block_size = 256
n_layer = 6
n_head = 6
n_embd = 384
dropout = 0.0
exec(open('configurator.py').read()) # overrides from command line or config file
# -----------------------------------------------------------------------------

torch.manual_seed(seed)
if torch.cuda.is_available():
    torch.cuda.manual_seed(seed)
torch.backends.cuda.matmul.allow_tf32 = True # allow tf32 on matmul
torch.backends.cudnn.allow_tf32 = True # allow tf32 on cudnn
device_type = 'cuda' if 'cuda' in device else 'cpu' # for later use in torch.autocast
ptdtype = {'float32': torch.float32, 'bfloat16': torch.bfloat16, 'float16': torch.float16}[dtype]
ctx = nullcontext() if device_type == 'cpu' else torch.amp.autocast(device_type=device_type, dtype=ptdtype)


"""
# Búsqueda de los pesos de entrenamiento.

**Autor:** Andrej Karpathy
**Documentación en español:** *Josué Aybar Paulino*

## Descripción

En esta condicional, se procede a la búsqueda de los pesos ya entrenados del modelo para cargarlo.
"""
# model
checkpoint = None
meta = None
meta_path = None
load_meta = False
if init_from == 'resume':
    # init from a model saved in a specific directory
    ckpt_path = os.path.join(out_dir, 'ckpt.pt')
    if not os.path.exists(ckpt_path):
        raise FileNotFoundError(
            f"No checkpoint found at {ckpt_path}. Train first or pass the correct --out_dir."
        )
    checkpoint = torch.load(ckpt_path, map_location=device)
    gptconf = GPTConfig(**checkpoint['model_args'])
    model = GPT(gptconf)
    state_dict = checkpoint['model']
    unwanted_prefix = '_orig_mod.'
    for k,v in list(state_dict.items()):
        if k.startswith(unwanted_prefix):
            state_dict[k[len(unwanted_prefix):]] = state_dict.pop(k)
    model.load_state_dict(state_dict)
    if 'config' in checkpoint and 'dataset' in checkpoint['config']: # older checkpoints might not have these...
        meta_path = os.path.join('data', checkpoint['config']['dataset'], 'meta.pkl')
        load_meta = os.path.exists(meta_path)
elif init_from == 'scratch':
    # init a new model with random weights so we can sample before training
    meta_path = os.path.join('data', dataset, 'meta.pkl')
    if not os.path.exists(meta_path):
        raise FileNotFoundError(
            f"No meta.pkl found at {meta_path}. Run data/{dataset}/prepare.py first "
            "or pass the correct --dataset."
        )
    load_meta = True
    with open(meta_path, 'rb') as f:
        meta = pickle.load(f)
    model_args = dict(
        n_layer=n_layer,
        n_head=n_head,
        n_embd=n_embd,
        block_size=block_size,
        bias=bias,
        vocab_size=meta['vocab_size'],
        dropout=dropout,
    )
    print(f"Initializing a new untrained model from scratch using {meta_path}")
    gptconf = GPTConfig(**model_args)
    model = GPT(gptconf)
elif init_from.startswith('gpt2'):
    # init from a given GPT-2 model
    model = GPT.from_pretrained(init_from, dict(dropout=0.0))
else:
    raise ValueError("init_from must be 'scratch', 'resume', or a GPT-2 variant such as 'gpt2'.")

model.eval()
model.to(device)
if compile:
    model = torch.compile(model) # requires PyTorch 2.0 (optional)

"""
# Carga del entrenamiento.

**Autor:** Andrej Karpathy
**Documentación en español:** *Josué Aybar Paulino*

## Descripción

Esta condicional es crucial pues procede a cargar todo el vocabulario de Shakespeare al modelo y declara 
las funciones de conversión que se utilizarán `stoi` e `itos`. 
"""
if load_meta:
    print(f"Loading meta from {meta_path}...")
    if meta is None:
        with open(meta_path, 'rb') as f:
            meta = pickle.load(f)
    # TODO want to make this more general to arbitrary encoder/decoder schemes
    stoi, itos = meta['stoi'], meta['itos']
    encode = lambda s: [stoi[c] for c in s]
    decode = lambda l: ''.join([itos[i] for i in l])


    """
    # Conversión de los datos.

    **Autor:** Andrej Karpathy
    **Documentación en español:** *Josué Aybar Paulino*

    ## Descripción

    Esta codicional ejecuta la converción de letras a números y números a letras. 
    """
else:
    # ok let's assume gpt-2 encodings by default
    print("No meta.pkl found, assuming GPT-2 encodings...")
    enc = tiktoken.get_encoding("gpt2")
    encode = lambda s: enc.encode(s, allowed_special={"<|endoftext|>"})
    decode = lambda l: enc.decode(l)


"""
# Organización de los datos de entrada.

**Autor:** Andrej Karpathy
**Documentación en español:** *Josué Aybar Paulino*

## Descripción

Esta función X = (torch, heredada de Pytorch, organiza los datos de stoi en secuencia para que el modelo
Transformer pueda procesarlos.
"""
# encode the beginning of the prompt
if start.startswith('FILE:'):
    with open(start[5:], 'r', encoding='utf-8') as f:
        start = f.read()
start_ids = encode(start)
x = (torch.tensor(start_ids, dtype=torch.long, device=device)[None, ...])


"""
# Generación de los texto.

**Autor:** Andrej Karpathy
**Documentación en español:** *Josué Aybar Paulino*

## Descripción

Parte final del recorrido donde luego de haber procesado los datos numéricos de entrada (x), 
de haber considerado la aleatoriedad del modelo (temperatura 0.8); así como de las 
representaciones de tokens más probables (Tok_k), finalmente, se genera el siguiente token que 
continúa la secuencia del texto proporcionado. Lo hace de uno en uno de forma autorregresiva hasta 
alcanzar el número máximo de tokens. 
"""
# run generation
with torch.no_grad():
    with ctx:
        for k in range(num_samples):
            y = model.generate(x, max_new_tokens, temperature=temperature, top_k=top_k)
            print(decode(y[0].tolist()))
            print('---------------')
